## Usage

```tsx
import { Message, MessageAvatar, MessageContent, MessageHeader, MessageFooter } from "uilab-core"
```

```tsx
<Message align="start">
  <MessageAvatar>👤</MessageAvatar>
  <MessageContent>
    <MessageHeader>Bat</MessageHeader>
    <Bubble>
      <BubbleContent>Hello!</BubbleContent>
    </Bubble>
    <MessageFooter>10:30 AM</MessageFooter>
  </MessageContent>
</Message>
```
