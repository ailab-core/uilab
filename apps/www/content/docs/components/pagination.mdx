## Usage
```tsx
import {
  Pagination,
  PaginationContent,
  PaginationEllipsis,
  PaginationItem,
  PaginationLink,
  PaginationNext,
  PaginationPrevious,
} from "uilab-core"
```
```tsx
<Pagination>
  <PaginationContent>
    <PaginationItem>
      <PaginationPrevious href="#" />
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#">1</PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#" isActive>
        2
      </PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#">3</PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationEllipsis />
    </PaginationItem>
    <PaginationItem>
      <PaginationNext href="#" />
    </PaginationItem>
  </PaginationContent>
</Pagination>
```

## TanStack Router
By default the `<PaginationLink />` component renders an `<a />` tag. Wrap it with `createLink` to get a type-safe router link with `to`, `params` and `search`.
```tsx
import { createLink } from "@tanstack/react-router"
import { PaginationLink } from "uilab-core"

const PaginationRouterLink = createLink(PaginationLink)

<PaginationRouterLink to="/posts" search={{ page: 2 }} isActive>
  2
</PaginationRouterLink>
```
