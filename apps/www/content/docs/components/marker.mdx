## Usage

```tsx
import { Marker, MarkerIcon, MarkerContent } from "uilab-core"
```

```tsx
<Marker>
  <MarkerIcon>
    <Clock />
  </MarkerIcon>
  <MarkerContent>10:30 AM</MarkerContent>
</Marker>
```
